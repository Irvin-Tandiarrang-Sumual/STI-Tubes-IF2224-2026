function Calculate(a: integer; b: integer): integer;
var
  result: integer;
  shadowVar: boolean; { EDGE CASE 2: Shadowing (Harus sukses) }
begin
  { Harusnya mengubah shadowVar lokal (boolean), bukan global (integer) }
  shadowVar := true; 
  result := a + b;
  
  { EDGE CASE 3: Type Mismatch dalam Assignment (Harus error) }
  { result := shadowVar; } 
  
  Calculate := result;
end;

{ Uji Prosedur dan Control Flow }
procedure TestControlFlow(limit: integer);
var
  i: integer;
begin
  for i := 1 to limit do
  begin
    if i > 5 then
      writeln('Greater', i)
    else
      writeln('Less or equal');
  end;
end;
