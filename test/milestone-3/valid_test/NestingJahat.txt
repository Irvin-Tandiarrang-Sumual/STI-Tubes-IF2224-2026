program DeepNest;

var
    a, b, c, d : integer;

begin
    for a := 1 to 2 do
    begin
        for b := 1 to 2 do
        begin
            while b < 3 do
            begin
                repeat
                    for c := 1 to 2 do
                    begin
                        for d := 1 to 2 do
                        begin
                            writeln(a, b, c, d);
                        end;
                    end;

                    b := b + 1;
                until b >= 3;
            end;
        end;
    end;
end.
